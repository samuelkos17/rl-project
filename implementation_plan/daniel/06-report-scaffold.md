>
> - **`confirms_h1` no longer means "the CI excludes zero".** Spec §1 states two
>   conditions and it now requires both: CI excluding zero **and** a correlation
>   that grows with difficulty. Printing it under the old label would state
>   something false about the project's headline result — a negative trend with a
>   positive CI would print "CI excludes zero: False". The half you want for that
>   label is the new `ci_excludes_zero` key. Fixed inline below.
> - **H2 has a function now: `compare_coverage_predictors(df, seed=0)`.** Spec §1
>   requires task-relevant coverage to beat raw *with non-overlapping CIs*. The
>   draft below prints both correlations and leaves the comparison to the
>   reader's eye. Use the function's `confirms_h2` / `cis_overlap` /
>   `task_minus_raw` instead, so the verdict is computed rather than eyeballed.
>
> Also available and worth putting in the results file: `performance_profile`
> (spec §7.2), and the per-instance `rho_ci_low` / `rho_ci_high` columns that
> `within_instance_correlation` now returns — `per.to_markdown()` picks them up

- [ ] **Step 2: Write `src/rlx/analysis/report.py`**

```python
"""Generate report/results.md with every number the report needs.

Run:  python -m rlx.analysis.report --results results --out report/results.md
"""

import argparse
from pathlib import Path

import pandas as pd

from rlx.analysis.aggregate import load_all
from rlx.analysis.stats import (
    aggregate_correlation, build_analysis_table, iqm_by_strategy, rank_stability,
    within_instance_correlation,
)


def _hypothesis_section(df: pd.DataFrame) -> str:
    lines = ["## Central test: does early coverage predict final return?", ""]
    for col, label in (("early_auc_raw", "Raw coverage"),
                       ("early_auc_task", "Task-relevant coverage")):
        per = within_instance_correlation(df, col)
        agg = aggregate_correlation(per)
        lines += [
            f"### {label}", "",
            f"- Mean within-instance Spearman rho: **{agg['mean_rho']:+.3f}**",
            f"- 95% bootstrap CI: [{agg['ci_low']:+.3f}, {agg['ci_high']:+.3f}]",
            f"- Instances with usable variance: {agg['n_instances']}",
            f"- Correlation vs difficulty trend: {agg['trend_with_difficulty']:+.3f} "
            f"(H1 predicts positive: stronger on harder mazes)",
            f"- CI excludes zero: {agg['ci_excludes_zero']}",
            f"- **H1 confirmed (CI excludes zero AND trend positive): "
            f"{agg['confirms_h1']}**", "",
            "Per instance:", "",
            per.to_markdown(index=False), "",
        ]
    return "\n".join(lines)


def _iqm_section(df: pd.DataFrame) -> str:
    rows = []
    for env_id in sorted(df["env_id"].unique()):
        for strategy, r in iqm_by_strategy(df, env_id).items():
            rows.append({"env_id": env_id, "strategy": strategy,
                         "iqm": round(r["iqm"], 3),
                         "ci": f"[{r['ci_low']:.3f}, {r['ci_high']:.3f}]",
                         "n_seeds": r["n"]})
    return ("## IQM final return by strategy and instance\n\n"
            + pd.DataFrame(rows).to_markdown(index=False) + "\n")


def build_report(results_root: Path, out_path: Path) -> None:
    runs = load_all(results_root)
    if not runs:
        raise SystemExit(f"no runs found under {results_root}")
    df = build_analysis_table(runs)

    winners = (df.groupby(["env_id", "strategy"])["final_return"].mean()
                 .reset_index()
                 .sort_values("final_return", ascending=False)
                 .groupby("env_id").first()
                 .reset_index()[["env_id", "strategy", "final_return"]])

    sections = [
        "# Results\n",
        "Generated by `python -m rlx.analysis.report`. Do not edit by hand.\n",
        f"- Runs analysed: **{len(df)}**",
        f"- Environment instances: **{df['env_id'].nunique()}**",
        f"- Strategies: **{df['strategy'].nunique()}**",
        f"- Seeds per configuration: **{df.groupby(['env_id', 'strategy']).size().max()}**\n",
        _hypothesis_section(df),
        _iqm_section(df),
        "## Rank stability (H3)\n",
        rank_stability(df).round(3).to_markdown(index=False) + "\n",
        "## Best strategy per instance\n",
        winners.round(3).to_markdown(index=False) + "\n",
        "## Full per-run table\n",
        df.round(4).to_markdown(index=False) + "\n",
    ]

    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_text("\n".join(sections), encoding="utf-8")


def main() -> None:
    p = argparse.ArgumentParser()
    p.add_argument("--results", type=Path, default=Path("results"))
    p.add_argument("--out", type=Path, default=Path("report/results.md"))
    args = p.parse_args()
    build_report(args.results, args.out)
    print(f"wrote {args.out}")


if __name__ == "__main__":
    main()
```

def test_report_generation_produces_every_section(synthetic, tmp_path):
    from rlx.analysis.report import build_report

    out = tmp_path / "results.md"
    build_report(synthetic, out)
    text = out.read_text(encoding="utf-8")
    for heading in ("# Results", "Central test", "IQM final return",
                    "Rank stability", "Best strategy per instance"):
        assert heading in text, heading
    assert len(text) > 2_000

Then read `report/results.md` before writing a word of prose. Two things to check
first:

- **Does `confirms_h1` say true or false?** If false, the report is not wrong —
  it just tells a different story, and the discussion section becomes about why
  coverage did not predict performance. Do not go looking for a slicing of the
  data that makes it true. That would be the fishing expedition §1 of the spec
  was written to prevent.
- **How many instances came out `NaN`?** If most of them did, nearly every run
  scored zero and the honest headline is "every method we tested fails beyond
  difficulty X" — a real finding, reported as such.
